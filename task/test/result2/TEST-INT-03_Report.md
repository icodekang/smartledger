```python
# 测试数据：金额1000元，税额130元，价税合计1130元

    {"subject_code": "1403", "debit": 1000.00, "credit": 0},      # 原材料
    {"subject_code": "222101", "debit": 130.00, "credit": 0},    # 进项税额
    {"subject_code": "2202", "debit": 0, "credit": 1130.00},     # 应付账款
]

total_debit = sum(e["debit"] for e in entries)   # 1130.00
total_credit = sum(e["credit"] for e in entries) # 1130.00

assert total_debit == total_credit  # ✅ 通过
```
