## 3. 技术实现

```typescript
// 与主系统共用API，但权限受限
// 只能访问自己的客户数据

// API权限控制示例
const customerAuthMiddleware = async (req, res, next) => {
  const token = req.headers.authorization
  const decoded = jwt.verify(token, SECRET)
  
  // 客户Portal用户
  if (decoded.type === 'customer') {
    req.customerId = decoded.customerId
    // 后续所有查询限制在该customerId下
  }
  
  next()
}
```

## 4. 权限控制

```python
# 客户只能访问自己的数据
def get_customer_data(customer_id: str, user: User):
    """获取客户数据（权限控制）"""
    
    # 检查用户是否属于该客户
    if user.customer_id != customer_id:
        raise HTTPException(403, "无权访问")
    
    return db.query(Customer).get(customer_id)


# API路由
@app.get("/api/v1/portal/customers/{id}/reports")
def get_customer_reports(id: str, user: PortalUser = Depends(get_current_user)):
    """客户查看自己的报表"""
    
    # 强制验证只能看自己的
    if str(user.customer_id) != id:
        raise HTTPException(403, "无权访问")
    
    return get_reports(customer_id=id)
```
